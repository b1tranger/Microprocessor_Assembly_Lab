### `Project.MM/ESP32_NAS_Server/src/main.cpp`
```cpp
#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <SPI.h>
#include <SD.h>

#define SD_CS_PIN 5

// Wokwi virtual Wi-Fi access point credentials
const char* ssid = "Wokwi-GUEST";
const char* password = "";

WebServer server(80);

String formatBytes(size_t bytes) {
  if (bytes < 1024) return String(bytes) + " B";
  if (bytes < (1024 * 1024)) return String(bytes / 1024.0, 1) + " KB";
  return String(bytes / (1024.0 * 1024.0), 1) + " MB";
}

void ensureInitialFiles() {
  if (!SD.exists("/www")) {
    SD.mkdir("/www");
  }

  if (!SD.exists("/www/index.html")) {
    File html = SD.open("/www/index.html", FILE_WRITE);
    if (html) {
      html.println("<!DOCTYPE html><html><head><meta charset='UTF-8'>");
      html.println("<title>ESP32 Web Server</title></head>");
      html.println("<body style='font-family:sans-serif; text-align:center; padding:30px; background:#f4f4f4;'>");
      html.println("<h1>Hosted on ESP32 MicroSD Storage</h1>");
      html.println("<p>Dual-Purpose Node: <strong>Static Web Host Active</strong></p>");
      html.println("<p><a href='/'>Go to NAS File Browser</a></p>");
      html.println("</body></html>");
      html.close();
      Serial.println("[STORAGE] Seeded /www/index.html");
    }
  }

  if (!SD.exists("/lab_report.txt")) {
    File report = SD.open("/lab_report.txt", FILE_WRITE);
    if (report) {
      report.println("Course: CSE0611324 Microprocessor Lab");
      report.println("Project: ESP32 Embedded NAS & Web Node");
      report.println("Storage: FAT32 block device over hardware SPI");
      report.close();
      Serial.println("[STORAGE] Seeded /lab_report.txt");
    }
  }
}

// NAS File Browser Route Handler
void handleRoot() {
  uint64_t total = SD.totalBytes();
  uint64_t used = SD.usedBytes();

  String html = "<!DOCTYPE html><html><head><meta charset='UTF-8'>";
  html += "<meta name='viewport' content='width=device-width, initial-scale=1.0'>";
  html += "<title>ESP32 NAS Node</title><style>";
  html += "body { font-family: monospace; background: #121212; color: #e0e0e0; padding: 20px; }";
  html += ".card { max-width: 650px; margin: auto; background: #1e1e1e; padding: 20px; border-radius: 8px; }";
  html += "table { width: 100%; border-collapse: collapse; margin-top: 15px; }";
  html += "th, td { border: 1px solid #333; padding: 8px; text-align: left; }";
  html += "th { background: #2a2a2a; }";
  html += "a { color: #64B5F6; text-decoration: none; }";
  html += "a:hover { text-decoration: underline; }";
  html += "</style></head><body><div class='card'>";
  html += "<h2>ESP32 Local Storage NAS</h2>";
  html += "<p><b>Capacity:</b> " + formatBytes(total) + " | <b>Used:</b> " + formatBytes(used) + "</p>";
  html += "<p><a href='/site'>View Hosted Static Website (/site)</a></p>";
  html += "<table><tr><th>File Name</th><th>Size</th><th>Action</th></tr>";

  File root = SD.open("/");
  File file = root.openNextFile();
  while (file) {
    if (!file.isDirectory()) {
      String fname = String(file.name());
      if (fname.startsWith("/")) fname = fname.substring(1);
      html += "<tr><td>" + fname + "</td>";
      html += "<td>" + formatBytes(file.size()) + "</td>";
      html += "<td><a href='/get?file=" + fname + "'>Download</a></td></tr>";
    }
    file = root.openNextFile();
  }

  html += "</table></div></body></html>";
  server.send(200, "text/html", html);
}

// Static Web Host Endpoint
void handleSite() {
  if (SD.exists("/www/index.html")) {
    File htmlFile = SD.open("/www/index.html", FILE_READ);
    server.streamFile(htmlFile, "text/html");
    htmlFile.close();
  } else {
    server.send(404, "text/plain", "Static Site Not Found");
  }
}

// NAS File Stream Endpoint
void handleDownload() {
  if (!server.hasArg("file")) {
    server.send(400, "text/plain", "Missing file parameter");
    return;
  }

  String filename = "/" + server.arg("file");
  if (!SD.exists(filename)) {
    server.send(404, "text/plain", "File Not Found");
    return;
  }

  File downloadFile = SD.open(filename, FILE_READ);
  server.sendHeader("Content-Disposition", "attachment; filename=\"" + server.arg("file") + "\"");
  server.streamFile(downloadFile, "application/octet-stream");
  downloadFile.close();
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n--- Starting ESP32 NAS & Web Node (VS Code Local) ---");

  // Mount SD Storage
  if (!SD.begin(SD_CS_PIN)) {
    Serial.println("[ERROR] MicroSD card initialization failed!");
    return;
  }
  Serial.printf("[STORAGE] Online. Total Space: %llu MB\n", SD.totalBytes() / (1024 * 1024));
  ensureInitialFiles();

  // Connect to Virtual Wi-Fi
  Serial.printf("[WIFI] Connecting to %s...", ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(200);
    Serial.print(".");
  }
  Serial.println("\n[WIFI] Connected!");
  Serial.printf("[NET] Node IP Address: %s\n", WiFi.localIP().toString().c_str());

  // Bind Routes
  server.on("/", HTTP_GET, handleRoot);
  server.on("/site", HTTP_GET, handleSite);
  server.on("/get", HTTP_GET, handleDownload);

  server.begin();
  Serial.println("[HTTP] Server listening on port 80.");
}

void loop() {
  server.handleClient();
}
```

## 3. Analysis & Key Insights

1. **Decoupled Local Simulation Toolchain**:
   - The Wokwi extension decouples code compilation from the simulation engine. PlatformIO manages toolchains (`espressif32`, ESP-IDF/Arduino framework) locally, producing standard `.bin` and `.elf` files.
   - `wokwi.toml` acts as the bridge connecting the compiled binary output (`.pio/build/esp32dev/firmware.bin`) directly to the Wokwi QEMU/Wasm emulator.
2. **Hardware SPI Bus Interfacing in Simulation**:
   - The virtual SPI pins on the ESP32 (`GPIO 18` = SCK, `GPIO 19` = MISO/DO, `GPIO 23` = MOSI/DI, `GPIO 5` = CS) simulate block-level I/O to a virtual FAT32 MicroSD card.
   - Files created with `SD.open(..., FILE_WRITE)` persist across standard runtime operations in memory.
3. **Local Loopback Network Forwarding (`wokwigw`)**:
   - Wokwi's virtual Wi-Fi (`Wokwi-GUEST`) runs on an isolated internal subnet (`10.13.37.x`).
   - Running `wokwigw.exe --forward 9080:10.13.37.2:80` punches a local bridge from the host machine's `localhost:9080` to the simulated ESP32 port 80, completely bypassing browser CORS and cloud paywalls.

```toml

```

---

## 3. Analysis & Key Insights

1. **Why `connect tcp 10.13.37.2:80: no route to host` Happened**:
   - `wokwigw.exe` runs a WebSocket server on `ws://localhost:9011` to accept simulation network traffic, while simultaneously running a TCP proxy (`tcpproxy`) forwarding host port 9080 to the simulated device IP `10.13.37.2:80`.
   - By default, without explicit configuration in `wokwi.toml`, Wokwi for VS Code does not connect to the local gateway WebSocket server at `localhost:9011`.
   - Because the simulation never registered as a client with `wokwigw`, `wokwigw`'s internal network routing table had no device at `10.13.37.2`. Thus, any HTTP request from the browser caused `tcpproxy` to fail immediately with `no route to host`.
2. **The Fix**:
   - Add the `[net]` configuration block in `wokwi.toml`:
     ```toml
     [net]
     gateway = "ws://localhost:9011"
     ```
   - Ensure `wokwigw.exe` is started **before** starting/restarting the Wokwi simulation in VS Code so that Wokwi establishes the WebSocket link on launch.

---

1. Updated `wokwi.toml` with `gateway = "ws://localhost:9011"`.
2. Updated `README.md` with detailed execution order and troubleshooting instructions.
3. Provided clear 4-step sequence to resolve the error and access `http://localhost:9080/`.

---

## 3. Analysis & Key Insights

1. **Compilation vs Simulation Separation**:
   - Wokwi does not compile C++ source files (`main.cpp`) into machine code. It expects the compiled `.bin` file to already exist at `.pio/build/esp32dev/firmware.bin`.
2. **PlatformIO Project Root Context**:
   - PlatformIO is folder-scoped: it requires opening the folder containing `platformio.ini` (`Project.MM/ESP32_NAS_Server/`) as the VS Code root workspace to activate the build toolchain and status bar icons.
   - Alternatively, running `pio run` inside `Project.MM/ESP32_NAS_Server/` compiles the binary directly.

---
