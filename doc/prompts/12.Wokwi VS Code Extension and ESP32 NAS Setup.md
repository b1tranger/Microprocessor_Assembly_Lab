### `Project.MM/ESP32_NAS_Server/src/main.cpp`
```cpp
#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <SPI.h>
#include <SD.h>

#define SD_CS_PIN 5

// Wokwi virtual Wi-Fi access point credentials
const char* ssid = "Wokwi-GUEST";
const char* password = "";

WebServer server(80);

String formatBytes(size_t bytes) {
  if (bytes < 1024) return String(bytes) + " B";
  if (bytes < (1024 * 1024)) return String(bytes / 1024.0, 1) + " KB";
  return String(bytes / (1024.0 * 1024.0), 1) + " MB";
}

void ensureInitialFiles() {
  if (!SD.exists("/www")) {
    SD.mkdir("/www");
  }

  if (!SD.exists("/www/index.html")) {
    File html = SD.open("/www/index.html", FILE_WRITE);
    if (html) {
      html.println("<!DOCTYPE html><html><head><meta charset='UTF-8'>");
      html.println("<title>ESP32 Web Server</title></head>");
      html.println("<body style='font-family:sans-serif; text-align:center; padding:30px; background:#f4f4f4;'>");
      html.println("<h1>Hosted on ESP32 MicroSD Storage</h1>");
      html.println("<p>Dual-Purpose Node: <strong>Static Web Host Active</strong></p>");
      html.println("<p><a href='/'>Go to NAS File Browser</a></p>");
      html.println("</body></html>");
      html.close();
      Serial.println("[STORAGE] Seeded /www/index.html");
    }
  }

  if (!SD.exists("/lab_report.txt")) {
    File report = SD.open("/lab_report.txt", FILE_WRITE);
    if (report) {
      report.println("Course: CSE0611324 Microprocessor Lab");
      report.println("Project: ESP32 Embedded NAS & Web Node");
      report.println("Storage: FAT32 block device over hardware SPI");
      report.close();
      Serial.println("[STORAGE] Seeded /lab_report.txt");
    }
  }
}

// NAS File Browser Route Handler
void handleRoot() {
  uint64_t total = SD.totalBytes();
  uint64_t used = SD.usedBytes();

  String html = "<!DOCTYPE html><html><head><meta charset='UTF-8'>";
  html += "<meta name='viewport' content='width=device-width, initial-scale=1.0'>";
  html += "<title>ESP32 NAS Node</title><style>";
  html += "body { font-family: monospace; background: #121212; color: #e0e0e0; padding: 20px; }";
  html += ".card { max-width: 650px; margin: auto; background: #1e1e1e; padding: 20px; border-radius: 8px; }";
  html += "table { width: 100%; border-collapse: collapse; margin-top: 15px; }";
  html += "th, td { border: 1px solid #333; padding: 8px; text-align: left; }";
  html += "th { background: #2a2a2a; }";
  html += "a { color: #64B5F6; text-decoration: none; }";
  html += "a:hover { text-decoration: underline; }";
  html += "</style></head><body><div class='card'>";
  html += "<h2>ESP32 Local Storage NAS</h2>";
  html += "<p><b>Capacity:</b> " + formatBytes(total) + " | <b>Used:</b> " + formatBytes(used) + "</p>";
  html += "<p><a href='/site'>View Hosted Static Website (/site)</a></p>";
  html += "<table><tr><th>File Name</th><th>Size</th><th>Action</th></tr>";

  File root = SD.open("/");
  File file = root.openNextFile();
  while (file) {
    if (!file.isDirectory()) {
      String fname = String(file.name());
      if (fname.startsWith("/")) fname = fname.substring(1);
      html += "<tr><td>" + fname + "</td>";
      html += "<td>" + formatBytes(file.size()) + "</td>";
      html += "<td><a href='/get?file=" + fname + "'>Download</a></td></tr>";
    }
    file = root.openNextFile();
  }

  html += "</table></div></body></html>";
  server.send(200, "text/html", html);
}

// Static Web Host Endpoint
void handleSite() {
  if (SD.exists("/www/index.html")) {
    File htmlFile = SD.open("/www/index.html", FILE_READ);
    server.streamFile(htmlFile, "text/html");
    htmlFile.close();
  } else {
    server.send(404, "text/plain", "Static Site Not Found");
  }
}

// NAS File Stream Endpoint
void handleDownload() {
  if (!server.hasArg("file")) {
    server.send(400, "text/plain", "Missing file parameter");
    return;
  }

  String filename = "/" + server.arg("file");
  if (!SD.exists(filename)) {
    server.send(404, "text/plain", "File Not Found");
    return;
  }

  File downloadFile = SD.open(filename, FILE_READ);
  server.sendHeader("Content-Disposition", "attachment; filename=\"" + server.arg("file") + "\"");
  server.streamFile(downloadFile, "application/octet-stream");
  downloadFile.close();
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n--- Starting ESP32 NAS & Web Node (VS Code Local) ---");

  // Mount SD Storage
  if (!SD.begin(SD_CS_PIN)) {
    Serial.println("[ERROR] MicroSD card initialization failed!");
    return;
  }
  Serial.printf("[STORAGE] Online. Total Space: %llu MB\n", SD.totalBytes() / (1024 * 1024));
  ensureInitialFiles();

  // Connect to Virtual Wi-Fi
  Serial.printf("[WIFI] Connecting to %s...", ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(200);
    Serial.print(".");
  }
  Serial.println("\n[WIFI] Connected!");
  Serial.printf("[NET] Node IP Address: %s\n", WiFi.localIP().toString().c_str());

  // Bind Routes
  server.on("/", HTTP_GET, handleRoot);
  server.on("/site", HTTP_GET, handleSite);
  server.on("/get", HTTP_GET, handleDownload);

  server.begin();
  Serial.println("[HTTP] Server listening on port 80.");
}

void loop() {
  server.handleClient();
}
```

## 3. Analysis & Key Insights

1. **Decoupled Local Simulation Toolchain**:
   - The Wokwi extension decouples code compilation from the simulation engine. PlatformIO manages toolchains (`espressif32`, ESP-IDF/Arduino framework) locally, producing standard `.bin` and `.elf` files.
   - `wokwi.toml` acts as the bridge connecting the compiled binary output (`.pio/build/esp32dev/firmware.bin`) directly to the Wokwi QEMU/Wasm emulator.
2. **Hardware SPI Bus Interfacing in Simulation**:
   - The virtual SPI pins on the ESP32 (`GPIO 18` = SCK, `GPIO 19` = MISO/DO, `GPIO 23` = MOSI/DI, `GPIO 5` = CS) simulate block-level I/O to a virtual FAT32 MicroSD card.
   - Files created with `SD.open(..., FILE_WRITE)` persist across standard runtime operations in memory.
3. **Local Loopback Network Forwarding (`wokwigw`)**:
   - Wokwi's virtual Wi-Fi (`Wokwi-GUEST`) runs on an isolated internal subnet (`10.13.37.x`).
   - Running `wokwigw.exe --forward 9080:10.13.37.2:80` punches a local bridge from the host machine's `localhost:9080` to the simulated ESP32 port 80, completely bypassing browser CORS and cloud paywalls.

## 3. Agent's Internal Thoughts & Reasoning Trail

- **Hypothesis 1 (Card Formatting / Driver Behavior)**: Investigated whether the Arduino-ESP32 v3.3 framework had breaking changes in `SD.begin()`. Inspected `SDFS::begin()` and `sdcard_init()` in `framework-arduinoespressif32/libraries/SD/src/SD.cpp` and `sd_diskio.cpp`. Discovered that `sdcard_init()` succeeds in registering the block device, but `ff_sd_initialize()` fails during the initial 74-clock cycle SPI power-up handshake (`transfer(0xFF)` with `card->ssPin`).
- **Hypothesis 2 (Wokwi Internal Netlist & Pin Identifiers)**: Since software logic was verified, examined how Wokwi routes signals to `wokwi-esp32-devkit-v1`. Searched Wokwi board definitions and discovered that the 30-pin devkit template labels digital GPIO pins with silkscreen prefixes: `D5`, `D18`, `D19`, `D23`.

In Wokwi's simulator netlist, these bare numbers do not exist as valid terminal pins on `wokwi-esp32-devkit-v1`. The engine was silently dropping all four SPI nets, resulting in zero electrical signals reaching the virtual card.
- **Resolution Path**: Changed all four endpoints to `"esp:D5"`, `"esp:D18"`, `"esp:D19"`, `"esp:D23"` in `diagram.json`, bound `SPI` explicitly in `SD.begin(SD_CS_PIN, SPI)`, and triggered incremental PlatformIO compilation.

---

## 4. Analysis & Key Insights

1. **Root Cause Discovery**:
   - The Wokwi component definition for `wokwi-esp32-devkit-v1` requires digital pins to be addressed by their printed silkscreen pin names (`D5`, `D18`, `D19`, `D23`).
   - Using bare numbers (`"esp:5"`, `"esp:18"`, `"esp:19"`, `"esp:23"`) caused Wokwi to consider all four SPI lines completely disconnected.
   - Consequently, the SPI bus had no physical signals reaching the virtual card, triggering `[PHASE 1 ERROR] SD Card initialization failed! Check SPI pins.`.
2. **Explicit SPI Bus Reference**:
   - Passing `SD.begin(SD_CS_PIN, SPI)` guarantees that the initialized hardware VSPI bus is bound directly to the filesystem driver.
3. **Compilation**:

The ESP32 runs a **32-bit Tensilica Xtensa Dual-Core CPU**.

Standard PC compilers (like Windows Visual C++, Clang, or GCC) can only generate **x86-64 machine code** that runs on Windows. If you sent x86-64 machine code to an ESP32 (real or simulated), the ESP32 CPU would crash instantly because it uses an entirely different instruction set architecture (ISA).

To compile code for an ESP32 on a Windows PC, you must use a **Cross-Compiler**: a compiler that runs on Windows (x86-64) but outputs machine instructions for Xtensa.

---

### 3. What Is Inside That ~600MB Download?

When PlatformIO runs for the first time on an ESP32 project, it downloads the complete embedded operating system and compiler suite into `~/.platformio/packages/`:

1. **`toolchain-xtensa-esp-elf` (~150MB)**:  
   The actual GNU C/C++ cross-compiler suite (`xtensa-esp32-elf-gcc`, `g++`, `ld`, `as`).
2. **`framework-arduinoespressif32` (~300MB)**:  
   The entire ESP32 Arduino Core and operating system, which includes:
   - **FreeRTOS kernel** (real-time operating system managing CPU cores and multitasking).
   - **LwIP TCP/IP stack** (networking stack managing sockets, IP addresses, and HTTP).
   - **ESP-IDF HAL drivers** (hardware abstraction layer for SPI, GPIO, timers, and Wi-Fi).
   - Built-in libraries: `WiFi`, `WebServer`, `HTTPClient`, `SD`, `FS`, and `SPI`.
3. **`framework-arduinoespressif32-esp32-libs` (~150MB)**:  
   Pre-compiled Espressif binary libraries (Wi-Fi blobs, crypto accelerators, Bluetooth controller).
4. **`tool-esptool`**:  
   The partition generator and flash image packager that bundles the bootloader, partition table, and application into `firmware.bin`.

---

### 4. "Why Doesn't Wokwi Web (in the browser) Download 600MB?"

When you use the online Wokwi browser simulator at `wokwi.com`:
- **Wokwi compiles your code on their remote Linux cloud servers.** Their cloud servers already have this ~600MB toolchain pre-installed.
- **The Trade-Off**: Because their servers pay for CPU and bandwidth, the web version places inbound HTTP networking behind a **paid subscription (Wokwi Club)**, rate-limits build times, and blocks direct `localhost` access to your computer's files.

**By running Wokwi locally in VS Code**:
- You bypass all cloud paywalls for free.
- You can access `http://localhost:9080` directly from your browser.
- **The only requirement**: Because compilation happens on **your** machine instead of Wokwi's cloud servers, your machine downloads the cross-compiler once.

---

### 5. Summary: How It All Fits Together

```
   [ main.cpp & html_templates.h ] (C++ Source Code)
                  │
                  ▼  (Executed by PlatformIO via ~600MB Cross-Compiler)
       [ firmware.bin & firmware.elf ] (Xtensa Machine Code)
                  │
                  ▼  (Loaded by wokwi.toml)
    ┌───────────────────────────┐
    │  Wokwi VS Code Extension  │  <─── Simulates ESP32 CPU, SPI, SD Card
    └─────────────┬─────────────┘
                  │  (Virtual Wi-Fi: 10.13.37.2:80)
                  ▼  (WebSocket: ws://localhost:9011)
         ┌─────────────────┐
         │  wokwigw.exe    │   <─── Bridges Network Packets
         └────────┬────────┘
                  │  (Forwarded to localhost:9080)
                  ▼
         [ Your PC Web Browser ]  (http://localhost:9080/)
```

- **PlatformIO**: Converts C++ $\rightarrow$ Binary (`firmware.bin`).
- **Wokwi Extension**: Executes the Binary inside a virtual ESP32 chip.
- **`wokwigw.exe`**: Connects your real PC browser to the simulated chip's virtual Wi-Fi.

Because this ~600MB toolchain is permanently cached in your user profile (`~/.platformio/packages/`), this download only happens **once**. Every build after that takes only **~8 seconds**.

2. **Self-Healing Automation**:

### 4. Critical Lab Edge Cases & Troubleshooting
